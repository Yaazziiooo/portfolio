03 / Behind the Code

Making the Shark Think

The shark uses a switch-based finite state machine with five states. It follows a spline until a player enters its detection range, then starts chasing and attacking.

When the target leaves the detection sphere, the shark looks for the closest available diver who isn't knocked down. If nobody is nearby, it returns to where it left its patrol route.

The shark's AI decisions run on the server, while movement and state are replicated to clients.

void AShark::Tick(float DeltaTime)
{
    Super::Tick(DeltaTime);
    if (!HasAuthority()) return;

    switch (CurrentState)
    {
        case ESharkState::Patrolling:
            MoveSpeed = PatrolSpeed;
            MoveAlongSpline(DeltaTime);
            if (IsValid(PlayerRef))
            {
                CurrentState = ESharkState::Chasing;
            }
            break;

        case ESharkState::Chasing:
            if (!IsValid(PlayerRef))
            {
                CurrentState = ESharkState::Returning;
                break;
            }

            if (FVector::Dist(GetActorLocation(), PlayerRef->GetActorLocation()) < AttackRange)
            {
                if (GetWorld()->GetTimeSeconds() >= RechargeTimestamp)
                    CurrentState = ESharkState::Attacking;

                break;
            }

            MoveSpeed = ChaseSpeed;
            MoveTowardPlayer(DeltaTime);
            break;

        case ESharkState::Attacking:
            PerformAttack();
            CurrentState = ESharkState::Recharging;
            break;

        case ESharkState::Returning:
            if (IsValid(PlayerRef))
            {
                CurrentState = ESharkState::Chasing;
                break;
            }

            ReturnToSpline(DeltaTime);
            MoveSpeed = PatrolSpeed;
            if (PatrolPath && PatrolPath->Spline)
            {
                FVector SplineLoc = PatrolPath->Spline->GetLocationAtDistanceAlongSpline(CurrentDistance, ESplineCoordinateSpace::World);
                if (FVector::Dist(GetActorLocation(), SplineLoc) < 200.f)
                {
                    CurrentState = ESharkState::Patrolling;
                }
            }
            break;

        case ESharkState::Recharging:
            if (GetWorld()->GetTimeSeconds() >= RechargeTimestamp)
                CurrentState = ESharkState::Chasing;
            break;
    }
}
04 / Behind the Code

Spawning Around the Propeller

When the submarine enters an algae area, algae spawn around its propeller.

I used a random height and a radius that changes with height to create an hourglass-shaped distribution. Taking the square root of the random distance keeps points spread evenly across each circle.

Spawn positions are calculated in the propeller's local space and transformed into world space, so they follow its position and rotation.

void AAlgaeSpawner::SpawnAlgae()
{
    if (!PropellerRef || !AlgaeClass) return;

    FVector SpawnCenterLocal(-128.342215f, -9.722936f, -83.838581f);
    float BaseRadius = 50.f;
    float MaxRadius  = 150.f;
    float Height     = 120.f;

    for (int32 i = 0; i < NumAlgaeToSpawn; i++)
    {
        if (FMath::FRand() < 0.5f)
            continue;

        float Z = FMath::FRandRange(-Height, Height);
        float CurrentRadius = FMath::Lerp(BaseRadius, MaxRadius, FMath::Clamp(FMath::Abs(Z) / Height, 0.f, 1.f));
        float Angle = FMath::FRandRange(0.f, 2 * PI);
        float Distance = FMath::Sqrt(FMath::FRand()) * CurrentRadius;

        float X = FMath::Cos(Angle) * Distance;
        float Y = FMath::Sin(Angle) * Distance;

        FVector RandomLocalOffset(X, Y, Z);
        FVector FinalLocalPos = SpawnCenterLocal + RandomLocalOffset;
        FVector WorldPos = PropellerRef->GetComponentTransform().TransformPosition(FinalLocalPos);
        FRotator RandomRot(0.f, FMath::FRandRange(0.f, 360.f), 0.f);

        AAlgae* Spawned = GetWorld()->SpawnActor<AAlgae>(AlgaeClass, WorldPos, RandomRot);
        if (Spawned)
        {
            Spawned->AttachToComponent(PropellerRef, FAttachmentTransformRules::KeepWorldTransform);
            SpawnedAlgae.Add(Spawned);
            Spawned->SetOwningSpawner(this);
        }
    }
}
